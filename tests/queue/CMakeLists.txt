cmake_minimum_required(VERSION 3.30)
project(AWJQueueContracts LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(awj_queue_merge_tests ../queue_merge_tests.cpp)
add_executable(awj_import_service_tests ../import_service_tests.cpp ../../src/ui/import_service.cpp)
if(WIN32)
    target_compile_definitions(awj_import_service_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
if(MSVC)
    target_compile_options(awj_queue_merge_tests PRIVATE /utf-8 /W4)
    target_compile_options(awj_import_service_tests PRIVATE /utf-8 /W4)
else()
    find_package(Threads REQUIRED)
    target_link_libraries(awj_import_service_tests PRIVATE Threads::Threads)
endif()
add_test(NAME queue_merge_contract COMMAND awj_queue_merge_tests)
add_test(NAME import_service_core COMMAND awj_import_service_tests)
set_tests_properties(queue_merge_contract import_service_core PROPERTIES TIMEOUT 30)
