cmake_minimum_required(VERSION 3.30)
project(AWJimageNodeApi LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "OHOS")
    message(FATAL_ERROR "此示例仅供真实 HarmonyOS Native SDK 构建")
endif()
if(NOT TARGET AWJimage::SDK)
    find_package(AWJimage 1.3 CONFIG REQUIRED)
endif()
find_library(AWJ_NAPI_LIBRARY NAMES ace_napi.z REQUIRED)
add_library(awj_napi SHARED awj_napi.cpp)
target_link_libraries(awj_napi PRIVATE AWJimage::SDK "${AWJ_NAPI_LIBRARY}")
target_compile_features(awj_napi PRIVATE cxx_std_17)
set_target_properties(awj_napi PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
