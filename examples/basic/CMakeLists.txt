cmake_minimum_required(VERSION 3.24)
project(AWJimageConsumers LANGUAGES C CXX)
find_package(AWJimage 1.3 CONFIG REQUIRED)
add_executable(awj-c-consumer basic.c)
target_link_libraries(awj-c-consumer PRIVATE AWJimage::SDK)
set_target_properties(awj-c-consumer PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
add_executable(awj-cxx-consumer basic.cpp)
target_link_libraries(awj-cxx-consumer PRIVATE AWJimage::SDK)
set_target_properties(awj-cxx-consumer PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
add_executable(awj-cxx17-consumer basic_cpp17.cpp)
target_link_libraries(awj-cxx17-consumer PRIVATE AWJimage::SDK)
target_compile_definitions(awj-cxx17-consumer PRIVATE AWJ_EXPECTED_VERSION="${AWJimage_VERSION}")
set_target_properties(awj-cxx17-consumer PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
if(WIN32)
    foreach(consumer IN ITEMS awj-c-consumer awj-cxx-consumer awj-cxx17-consumer)
        add_custom_command(TARGET ${consumer} POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:AWJimage::SDK>" "$<TARGET_FILE_DIR:${consumer}>" VERBATIM)
    endforeach()
endif()
enable_testing()
add_test(NAME installed_c_consumer COMMAND awj-c-consumer)
add_test(NAME installed_cxx_consumer COMMAND awj-cxx-consumer)
add_test(NAME installed_cxx17_consumer COMMAND awj-cxx17-consumer)
