cmake_minimum_required(VERSION 3.30)
# 由真实华为 HarmonyOS Native SDK 的 toolchain 配置 sysroot、架构与 API。
# 不在这里猜测 SDK 路径，不把 OpenHarmony/Linux 的产物当作 HarmonyOS 产物。
if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
    message(FATAL_ERROR "请显式指定华为 HarmonyOS Native SDK 的 CMAKE_TOOLCHAIN_FILE")
endif()
project(AWJimageHarmonyOS LANGUAGES C CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "OHOS")
    message(FATAL_ERROR "需要 HarmonyOS SDK 的 OHOS 目标；不能使用桌面宿主工具链")
endif()
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR CMAKE_CXX_COMPILER_VERSION VERSION_LESS 21)
    message(FATAL_ERROR "此项目要求经过真实 SDK 配套验证的 Clang 21+；不能只替换 SDK 编译器可执行文件")
endif()
set(AWJ_SDK_D3D11 OFF CACHE BOOL "" FORCE)
set(AWJ_SDK_SHM OFF CACHE BOOL "" FORCE)
set(AWJ_ENABLE_GPU_METRICS OFF CACHE BOOL "" FORCE)
set(AWJ_ENABLE_X64_V3 OFF CACHE BOOL "" FORCE)
set(AWJ_SDK_TOOLS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
# native codecs 默认开启；依赖必须已为目标 SDK/ABI 构建，不能使用 Windows vcpkg 库。
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/.." "${CMAKE_CURRENT_BINARY_DIR}/sdk")
option(AWJ_HARMONY_NODE_API "构建用于 ArkTS 的薄 Node-API 示例" OFF)
if(AWJ_HARMONY_NODE_API)
    add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../examples/harmonyos" "${CMAKE_CURRENT_BINARY_DIR}/napi")
endif()
