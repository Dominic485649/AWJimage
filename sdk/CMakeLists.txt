cmake_minimum_required(VERSION 3.30)
file(STRINGS "${CMAKE_CURRENT_LIST_DIR}/../VERSION" AWJ_SDK_VERSION)
project(AWJimageSDK VERSION ${AWJ_SDK_VERSION} LANGUAGES C CXX)
include(CTest)
option(AWJ_SDK_NATIVE_CODECS "包含项目锁定的原生图像编码器" ON)
set(AWJ_SDK_SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/..")
if(AWJ_SDK_NATIVE_CODECS)
    set(AWJ_BUILD_APPLICATION OFF CACHE BOOL "" FORCE)
    set(AWJ_BUILD_SDK ON CACHE BOOL "" FORCE)
    if(NOT DEFINED AWJ_ENABLE_GPU_METRICS)
        set(AWJ_ENABLE_GPU_METRICS OFF CACHE BOOL "" FORCE)
    endif()
    add_subdirectory("${AWJ_SDK_SOURCE_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/native")
else()
    include("${AWJ_SDK_SOURCE_ROOT}/cmake/sdk-targets.cmake")
endif()
